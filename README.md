# ebotLib
This is a tiny library of custom-view like Buttons, Slider and more soon



-- Include maven repositori in the "setting.gradle" files

repositories {
    ...
    maven { url = uri("https://jitpack.io") }
}

-- add library in the dependencies app's gradle files

dependencies {
    ...
    implementation("com.github.EmilioBoti:ebotLib:v0.0.1")
}
